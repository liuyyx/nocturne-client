# 成员 / 类的别名链 —— 生成期回退，不是需求面。
#
# 生成器先按 canonical 名字解析；解析不到时按顺序尝试右边的名字，命中即用。
# 表键始终是左边那个 canonical 名，所以运行期代码不用改。
#
# 格式（一行一条，`#` 之后是注释）:
#   ClassType常量#member()  = 备选1(), 备选2()     成员别名（方法带 ()，字段不带）
#   ClassType常量           = 备选类名             类别名（同一类的历史改名）
#
# 只在"映射源里这个成员换了名字"时用；来源缺失/权限问题不写在这里。

# 附魔：1.13–1.20.1 叫 getEnchantmentTags，1.8.9–1.12.2 叫 getEnchantmentTagList，
# 1.20.5+ 才是 getEnchantments（返回组件）。
ITEM_STACK#getEnchantments() = getEnchantmentTags(), getEnchantmentTagList()

# 跳跃冷却：1.8.9 是 jumpTicks，1.9+ 改名 noJumpDelay。
LIVING_ENTITY#jumpTicks = noJumpDelay

# 重生：1.8.9 只有 respawnPlayer，1.9+ 是 respawn。
LOCAL_PLAYER#respawnPlayer() = respawn()

# 时间：1.8.9–1.12.2 的 LevelData 只有 setWorldTime；1.13+ 才是 setDayTime。
# （canonical 侧是 1.13+ 名字，旧版回退到 setWorldTime。）
net/minecraft/world/level/storage/LevelData#setDayTime() = setWorldTime()

# 屏类：1.8.9–1.12.2 是 GuiScreen，1.13+ 是 Screen（同一层级的改名）。
SCREEN = GuiScreen

# 文本组件：1.19+ TextComponent 并入 MutableComponent（同一层级的改名）。
TEXT_COMPONENT = MUTABLE_COMPONENT

# 插值计时器：1.13 起 Timer 从 util 移到 client 包，1.21 改名 DeltaTracker。
# 表键仍是 DeltaTracker，运行期 `readField(..., DELTA_TRACKER, "partialTick")` 不用改。
net/minecraft/client/DeltaTracker = net/minecraft/client/Timer
# 1.21+ 的计时器实现是 DeltaTracker 的内部类；1.16.5–1.20.1 只有外层的 Timer。
# 运行期 DELTA_TIMER 用同一个字段名读写 msPerTick/tickLength，表键保持内部类名。
net/minecraft/client/DeltaTracker$Timer = net/minecraft/client/Timer
# 1.8.9–1.12.2 的等价字段叫 renderPartialTicks（1.13 起才改名 partialTick）。
net/minecraft/client/DeltaTracker#partialTick = renderPartialTicks
net/minecraft/client/Timer#partialTick = renderPartialTicks

# 26.x 去掉了 get 前缀：Camera#getPosition/getYRot/getXRot/getEntity 改名
# position()/yRot()/xRot()/entity()，GameRenderer#getMainCamera 改名 mainCamera()。
net/minecraft/client/Camera#getPosition() = position()
net/minecraft/client/Camera#getYRot() = yRot()
net/minecraft/client/Camera#getXRot() = xRot()
net/minecraft/client/Camera#getEntity() = entity()
net/minecraft/client/renderer/GameRenderer#getMainCamera() = mainCamera()
# 1.8.9–1.12.2 的有效 FOV 方法在 EntityRenderer 上叫 getFOVModifier(F,Z)。
net/minecraft/client/renderer/GameRenderer#getFov() = getFOVModifier()

# 目标分类：这些基类在 1.14–1.21.11 的 Mojmap 路径与最新规范名不同。
